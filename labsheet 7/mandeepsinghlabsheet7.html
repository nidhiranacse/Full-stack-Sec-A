# models.py
from django.db import models

class Fruit(models.Model):
    name = models.CharField(max_length=100)
    category = models.CharField(max_length=50)
    stock = models.IntegerField(default=0)

class SelectedEventStudent(models.Model):
    student_id = models.CharField(max_length=20, unique=True)
    name = models.CharField(max_length=100)
    event_name = models.CharField(max_length=100)
    rank_index = models.PositiveIntegerField()

# views.py
from django.shortcuts import render
from django.db.models import Q

def dashboard_view(request):
    search_query = request.GET.get('q', '').strip()
    sort_by = request.GET.get('sort', 'rank_index') # Default sort order

    # Fetch and filter fruits collection
    fruits = Fruit.objects.all()
    if search_query:
        fruits = fruits.filter(Q(name__icontains=search_query) | Q(category__icontains=search_query))

    # Fetch, filter, and dynamically order student matrix (Task 8.2 & 8.3)
    students = SelectedEventStudent.objects.all()
    if search_query:
        students = students.filter(
            Q(name__icontains=search_query) | 
            Q(event_name__icontains=search_query) | 
            Q(student_id__icontains=search_query)
        )
    
    # Toggle sorting direction or field properties
    allowed_sorts = ['name', '-name', 'rank_index', '-rank_index', 'event_name', '-event_name']
    if sort_by in allowed_sorts:
        students = students.order_by(sort_by)

    context = {
        'fruits': fruits,
        'students': students,
        'search_query': search_query,
        'current_sort': sort_by,
    }
    return render(request, 'dashboard.html', context)
