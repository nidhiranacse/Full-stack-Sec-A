const express = require('express');
const jwt = require('jsonwebtoken');

const app = express();
app.use(express.json());

const JWT_SECRET = 'your-256-bit-secret-key-here';

// In-memory database array
const users = [];

// ========================================================
// 1. POST /register
// ========================================================
app.post('/register', (req, res) => {
  const { username, password } = req.body;

  // Validation: username missing/empty OR password < 6 characters
  if (!username || typeof username !== 'string' || !username.trim() || !password || password.length < 6) {
    return res.status(400).json({ error: "Invalid input" });
  }

  // Check if username already taken
  const existingUser = users.find(u => u.username === username);
  if (existingUser) {
    return res.status(409).json({ error: "Username already exists" });
  }

  // Save new user
  users.push({ username, password });
  return res.status(201).json({ message: "User registered" });
});

// ========================================================
// 2. POST /login
// ========================================================
app.post('/login', (req, res) => {
  const { username, password } = req.body;

  // Find user and verify password
  const user = users.find(u => u.username === username && u.password === password);
  if (!user) {
    return res.status(401).json({ error: "Invalid credentials" });
  }

  // Sign JWT token containing username payload expiring in 1 hour
  const token = jwt.sign({ username: user.username }, JWT_SECRET, { expiresIn: '1h' });

  return res.status(200).json({ token });
});

// ========================================================
// 3. GET /profile
// ========================================================
app.get('/profile', (req, res) => {
  const authHeader = req.headers['authorization'];

  if (!authHeader || !authHeader.startsWith('Bearer ')) {
    return res.status(401).json({ error: "Unauthorized" });
  }

  const token = authHeader.split(' ')[1];

  try {
    // Verify token signature and expiration
    const decoded = jwt.verify(token, JWT_SECRET);
    return res.status(200).json({ username: decoded.username });
  } catch (err) {
    // Header missing, malformed, invalid signature or expired token
    return res.status(401).json({ error: "Unauthorized" });
  }
});

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
  console.log(`Server listening on http://localhost:${PORT}`);
});
