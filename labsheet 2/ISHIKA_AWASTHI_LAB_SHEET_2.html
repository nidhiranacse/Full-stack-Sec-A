A landing surface that rewires itself in one click.

Every color on this page is a CSS custom property. Flipping the switch above swaps a single data-theme attribute on <html> — no reload, no flash, no separate stylesheet.

const root = document.documentElement;
function toggleTheme() {
const next = root.dataset.theme === 'dark' ? 'light' : 'dark';
root.dataset.theme = next;
localStorage.setItem('wb-theme', next);
}
